% ============================================================
% Maquette du livre — structure.tex
% Corps en EB Garamond, composition « livre » à la française :
% alinéas, pas de saut de paragraphe, titres composés, veuves
% et orphelines contrôlées.
% ============================================================

% Format du livre : « a5 » (défaut) ou « poche », défini par le fichier
% maître (main.tex / main_poche.tex) avant l'\input de structure.tex
\usepackage{etoolbox}
\providecommand{\bookformat}{a5}

% Configuration des marges selon le format
\ifdefstring{\bookformat}{poche}{%
  \usepackage[paperwidth=108mm, paperheight=175mm, top=15mm, bottom=17mm, inner=17mm, outer=12mm, footskip=9mm]{geometry}%
  % Tailles de la page de titre intérieure (corps réduit pour le poche)
  \newcommand{\titlepageBig}{26}\newcommand{\titlepageMid}{19}%
}{%
  \usepackage[a5paper, top=20mm, bottom=22mm, inner=19mm, outer=16mm, footskip=11mm]{geometry}%
  \newcommand{\titlepageBig}{40}\newcommand{\titlepageMid}{30}%
}

% Table des matières et liens
\usepackage{tocloft, hyperref}
\hypersetup{
    pdfauthor={Kevin Delfour},
    pdftitle={L'envers du décor des entreprises libérées},
    pdfkeywords={entreprise libérée, holacratie, management, autonomie, organisation},
    hidelinks
}
\usepackage{xurl}    % autorise la césure des URL n'importe où
\urlstyle{same}      % URL composées dans la police du texte

% Cadres, tailles de police et en-têtes/pieds de page personnalisés
\usepackage{framed, anyfontsize, fancyhdr}
\usepackage{emptypage} % pages blanches (fin de chapitre) sans en-tête ni folio
\usepackage{lettrine}  % lettrines d'ouverture de chapitre
\setlength{\DefaultNindent}{0pt}
\renewcommand{\LettrineFontHook}{\bfseries}

% Police EB Garamond (corps du texte) et paramètres de langue
\usepackage{ebgaramond, float}
\usepackage[french]{babel}
\usepackage[T1]{fontenc}

% Couleurs personnalisées et graphiques TikZ
\usepackage[dvipsnames]{xcolor}
\usepackage{tikz, microtype, GS1, qrcode}
\usetikzlibrary{calc}

\usepackage[labelformat=empty]{caption}
\captionsetup{justification=raggedleft, font={tiny, color=gray}, singlelinecheck=false}

% Boucles, images et commandes supplémentaires
\usepackage{pgffor, graphicx}

% ------------------------------------------------------------
% Composition du texte : conventions du livre imprimé
% ------------------------------------------------------------
% Corps en Garamond (la police par défaut du document est la
% romane chargée ci-dessus) ; léger interlignage pour compenser
% la petite hauteur d'x de la Garamond.
\linespread{1.06}

% Paragraphes à la française : alinéa, pas de blanc entre les
% paragraphes (le blanc est réservé aux changements de section).
\setlength{\parindent}{1.2em}
\setlength{\parskip}{0pt plus 1pt}

% Veuves, orphelines et débordements
\clubpenalty=10000
\widowpenalty=10000
\displaywidowpenalty=10000
\ifdefstring{\bookformat}{poche}{\emergencystretch=2em}{\emergencystretch=1em}
\raggedbottom % pas de justification verticale : les pages courtes (intros de
              % partie, fins de chapitre) restent calées en haut au lieu d'être étirées

% ------------------------------------------------------------
% Titres (parties, chapitres, sections)
% ------------------------------------------------------------
\usepackage{titlesec}

% Chapitre : « CHAPITRE N » en petites capitales espacées et
% grisées, puis le titre en grand corps gras.
\titleformat{\chapter}[display]
  {\normalfont\raggedright}
  {\color{gray}\small\textls[120]{\MakeUppercase{\chaptertitlename~\thechapter}}}
  {0.6em}
  {\huge\bfseries}
\titlespacing*{\chapter}{0pt}{0pt}{2\baselineskip}

% Partie : numéro discret, titre en très grand corps entre deux filets centrés
\titleformat{\part}[display]
  {\centering\normalfont}
  {\color{gray}\large\scshape\textls[120]{\partname}\\[1.2em]\rule{0.45\textwidth}{0.4pt}}
  {1em}
  {\Huge\bfseries}
  [\vspace{1.2em}\rule{0.45\textwidth}{0.4pt}]

\titleformat{\section}
  {\normalfont\Large\bfseries}{\thesection}{0.6em}{}
\titlespacing*{\section}{0pt}{1.6\baselineskip}{0.6\baselineskip}

\titleformat{\subsection}
  {\normalfont\large\bfseries}{\thesubsection}{0.6em}{}
\titlespacing*{\subsection}{0pt}{1.2\baselineskip}{0.4\baselineskip}

\titleformat{\subsubsection}
  {\normalfont\normalsize\bfseries\itshape}{\thesubsubsection}{0.6em}{}

% Configuration de la table des matières
\setcounter{tocdepth}{2}
\renewcommand{\cftpartfont}{\normalfont\bfseries\Large}
\renewcommand{\cftchapfont}{\normalfont\bfseries}
\renewcommand{\cftsecfont}{\normalfont}
\renewcommand{\cftsubsecfont}{\normalfont}
\renewcommand{\cftsubsubsecfont}{\normalfont\small}

% En-têtes et pieds de page (recto-verso) :
% verso = titre du livre, recto = chapitre courant ; folio à l'extérieur
\pagestyle{fancy}
\fancyhead{}
\fancyhead[CE]{\color{gray}\scriptsize\scshape L'envers du décor des entreprises libérées}
\fancyhead[CO]{\color{gray}\scriptsize\scshape\leftmark}
\fancyfoot{}
\fancyfoot[LE]{\scriptsize\thepage}
\fancyfoot[RO]{\scriptsize\thepage}
\renewcommand{\headrulewidth}{0pt}

% ------------------------------------------------------------
% Liste des sources (notes de bas de page collectées)
% ------------------------------------------------------------
\newcommand{\listfootnotesname}{Liste des sources}
\newlistof[chapter]{footnotes}{fnt}{\listfootnotesname}
\setlength{\cftfootnotesnumwidth}{3.4em} % place pour « IV.12 »
\setlength{\cftfootnotesindent}{0pt}
\renewcommand{\cftfootnotesfont}{\small}
\renewcommand{\cftfootnotespagefont}{\small}

\newcommand{\insertImage}[2]{
    \markboth{}{} % Supprime les marques de tête
    \begin{figure}[H]
        \center
        % 0.92\textheight : laisse la place de la légende sous l'image
        \includegraphics[keepaspectratio, width=\textwidth, height=0.92\textheight]{images/#1}
        \captionsetup{justification=raggedright, singlelinecheck=false}
        \caption{#2}
    \end{figure}
    \newpage
}

\makeatletter
\renewcommand{\thefigure}{\@arabic\c@figure}
\makeatother

\let\oldfootnote\footnote % Save the old \footnote{...} command
\renewcommand\footnote[1]{%
    \oldfootnote{#1}% Make a regular footnote with added margin.
    \refstepcounter{footnotes}% Numérotation séquentielle des entrées de la liste
    \ifnum\value{part}>0
        \addcontentsline{fnt}{footnotes}{\protect\numberline{\thepart.\thefootnotes}#1}%
    \else % Avant la première partie (introduction) : pas de préfixe de partie
        \addcontentsline{fnt}{footnotes}{\protect\numberline{\arabic{footnotes}}#1}%
    \fi
}

\makeatletter
\@addtoreset{chapter}{part} % Réinitialise le compteur de chapitres à chaque nouvelle partie
\makeatother
